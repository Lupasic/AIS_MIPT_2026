\begin{frame}[t]{Research vs.\ Production Inference}
\framesubtitle{Why can't we just \texttt{model.predict()}?}
\begin{columns}[T,onlytextwidth]
    \begin{column}{0.49\textwidth}
        \textbf{Research / Notebook}
        \begin{itemize}
            \item One request at a time
            \item No latency SLA
            \item GPU underutilized between calls
            \item Framework-specific code everywhere
        \end{itemize}
    \end{column}
    \begin{column}{0.49\textwidth}
        \textbf{Production Server}
        \begin{itemize}
            \item Hundreds of concurrent users
            \item P99 latency $<$ 100\,ms
            \item GPU must stay busy
            \item Hardware-optimized execution graph
        \end{itemize}
    \end{column}
\end{columns}
\vspace{3mm}
\dbox{A production inference stack decouples the model from the serving logic and optimizes each layer independently.}
\end{frame}

\begin{frame}[t]{What Is ONNX?}
\framesubtitle{Open Neural Network Exchange}
\vspace*{-0.4cm}
\textbf{ONNX} is a hardware-agnostic, open format for storing trained neural network graphs.
\vspace{2mm}
\begin{tabular}{@{}p{3.2cm}p{8.0cm}@{}}
\toprule
Feature & Meaning \\
\midrule
Portable & Run the same \texttt{.onnx} file with PyTorch, TensorRT, OpenVINO, CoreML\ldots \\
Graph-based & Operators (Conv, ReLU, BN) are stored as a static computation graph \\
Runtime-agnostic & No Python, no PyTorch dependency on the server \\
Optimizable & Graph-level optimizations before hardware compilation \\
\bottomrule
\end{tabular}

{ONNX is the \textit{lingua franca} of deep learning inference.
Export once --- deploy anywhere.}
\end{frame}

\begin{frame}[t,fragile]{Exporting ResNet-152 to ONNX}
\vspace{-0.6cm}
\begin{lstlisting}[language=Python, basicstyle=\footnotesize\ttfamily]
# 1. Load model and append Softmax
backbone = models.resnet152(weights="DEFAULT")
backbone.eval()
model = torch.nn.Sequential(backbone, torch.nn.Softmax(dim=1))
# 2. Dummy input (any valid shape; used for tracing)
dummy = torch.randn(1, 3, 224, 224)
# 3. Export
torch.onnx.export(
    model, dummy, "resnet152.onnx",
    input_names=["input"],
    output_names=["output"],
    dynamic_axes={          # allow variable batch size
        "input":  {0: "batch_size"},
        "output": {0: "batch_size"},},
    opset_version=17,)
\end{lstlisting}
\end{frame}

\begin{frame}[t]{Why Dynamic Axes?}
\framesubtitle{Making the model batch-size agnostic}
Without \texttt{dynamic\_axes} the ONNX graph has a \textit{fixed} batch dimension of 1.
\vspace{2mm}
\begin{itemize}
    \item Triton and TensorRT need to send batches larger than 1 to saturate the GPU.
    \item A fixed-shape model will reject any input tensor whose batch size $\neq 1$.
    \item Dynamic axes mark axis 0 of \texttt{input} and \texttt{output} as symbolic
          (\texttt{"batch\_size"}), so the runtime can reshape them freely.
\end{itemize}
\vspace{2mm}
\dbox{Always set \texttt{dynamic\_axes} for at least the batch dimension when targeting a production server.}
\end{frame}

\begin{frame}[t]{Why Add Softmax?}
\framesubtitle{Convenience modification before export}
\begin{itemize}
    \item ResNet's last layer produces raw \textbf{logits} --- un-normalized scores.
    \item Downstream code then needs to apply \texttt{softmax} itself every time.
    \item By appending \texttt{torch.nn.Softmax(dim=1)} \textit{before} export, the ONNX
          model outputs class \textbf{probabilities} directly.
    \item The FastAPI response can read \texttt{argmax} straight from the Triton output.
\end{itemize}
\vspace{2mm}
\textbf{Rule of thumb:} bake post-processing into the graph whenever the extra operation is cheap and always applied.
\end{frame}

\begin{frame}[t]{NVIDIA Triton Inference Server}
\framesubtitle{What it is and what it solves}
Triton is a production-grade, open-source inference serving system by NVIDIA.
\vspace{2mm}
\begin{tabular}{@{}p{3.4cm}p{8.0cm}@{}}
\toprule
Capability & Details \\
\midrule
Multi-framework & ONNX Runtime, TensorRT, TensorFlow, PyTorch \\
Dynamic Batching & Aggregates concurrent single requests into one GPU batch \\
Instance Groups & Runs $N$ copies of a model in parallel on one or more GPUs \\
HTTP + gRPC & Clients can use either protocol \\
Metrics & Exposes latency, throughput, GPU utilization\\
\bottomrule
\end{tabular}
\end{frame}

\begin{frame}[t,fragile]{Triton: Model Repository Structure}
\framesubtitle{The required folder layout}
\vspace*{-0.4cm}
Triton expects a strict directory hierarchy:
\begin{lstlisting}[language=bash, basicstyle=\footnotesize\ttfamily]
models/
  resnet152/           # model name (used in API calls)
    config.pbtxt       # model configuration file
    1/                 # version number (integer folder)
      model.onnx       # the model file
\end{lstlisting}
\vspace{2mm}
Key rules:
\begin{itemize}
    \item The version folder \textit{must} be a number (\texttt{1}, \texttt{2}, \ldots).
    \item The file inside must be named \texttt{model.onnx} (or \texttt{model.plan} for TensorRT).
    \item Multiple versions can coexist; Triton serves the latest by default.
\end{itemize}
\end{frame}


\begin{frame}[t]{FastAPI Wrapper: Architecture}
\framesubtitle{Why an extra API layer?}
\begin{columns}[T,onlytextwidth]
    \begin{column}{0.49\textwidth}
        \textbf{Triton speaks tensors.}\\
        It expects raw FP32 arrays with exact shapes.
        \vspace{3mm}

        The user sends a \texttt{.jpg} file.
        Someone must bridge the gap.
    \end{column}
    \begin{column}{0.49\textwidth}
        \textbf{FastAPI service responsibilities:}
        \begin{enumerate}
            \item Accept multipart image upload
            \item Run preprocessing pipeline
            \item Call Triton via \texttt{tritonclient}
            \item Return JSON with top class
        \end{enumerate}
    \end{column}
\end{columns}
\vspace{3mm}
\dbox{The FastAPI layer is \textit{thin}: no model logic, no weights. Preprocessing + glue code only.}
\end{frame}

\begin{frame}[t,fragile]{FastAPI + tritonclient: gRPC Inference}
\vspace*{-0.5cm}
\begin{lstlisting}[language=Python, basicstyle=\footnotesize\ttfamily]
async def triton_infer(image_np: np.ndarray) -> np.ndarray:
    client = grpcclient.InferenceServerClient("localhost:8001")
    inp = grpcclient.InferInput("input", image_np.shape, "FP32")
    inp.set_data_from_numpy(image_np)
    out = grpcclient.InferRequestedOutput("output")
    response = await client.infer(
        model_name="resnet152",
        inputs=[inp],
        outputs=[out],)
    return response.as_numpy("output")   # shape: (1, 1000)
\end{lstlisting}
\textbf{gRPC vs HTTP}: gRPC serialization is $\sim$2$\times$ faster for large tensors; use it whenever the client supports it.
\end{frame}

\begin{frame}[t]{TensorRT: What and Why}
\framesubtitle{Hardware-specific compilation of the inference graph}
\textbf{TensorRT} compiles an ONNX model into a hardware-optimized \texttt{.plan} engine.
\vspace{2mm}
\begin{tabular}{@{}p{3.0cm}p{8.2cm}@{}}
\toprule
Optimization & Effect \\
\midrule
Layer fusion & Merges Conv+BN+ReLU into a single CUDA kernel \\
Kernel autotuning & Selects the fastest GEMM algorithm for the exact GPU \\
FP16 / INT8 & Halves or quarters memory bandwidth; faster tensor core ops \\
Memory pooling & Pre-allocates GPU buffers, removes allocation overhead \\
\bottomrule
\end{tabular}

A TensorRT-compiled ResNet-152 in FP16 runs \textbf{2--4$\times$ faster} than ONNX Runtime on the same GPU.
\end{frame}


\begin{frame}[t,fragile]{Serving TensorRT with Triton}
\framesubtitle{Switching from ONNX to TensorRT engine --- only two changes}
\begin{enumerate}
    \item Place \texttt{model.plan} in the version folder instead of \texttt{model.onnx}.
    \item Update \texttt{config.pbtxt}:
\end{enumerate}
\vspace{2mm}
\begin{columns}[T,onlytextwidth]
    \begin{column}{0.49\textwidth}
        \textbf{ONNX Runtime}
\begin{lstlisting}[basicstyle=\footnotesize\ttfamily]
platform: "onnxruntime_onnx"
\end{lstlisting}
    \end{column}
    \begin{column}{0.49\textwidth}
        \textbf{TensorRT}
\begin{lstlisting}[basicstyle=\footnotesize\ttfamily]
platform: "tensorrt_plan"
\end{lstlisting}
    \end{column}
\end{columns}
\vspace{2mm}
Everything else (FastAPI, tritonclient calls, monitoring) stays identical.
\dbox{The engine file is GPU-specific. Recompile with \texttt{trtexec} when moving to a different GPU model.}
\end{frame}

\begin{frame}[t]{Monitoring: Prometheus + Grafana}
\framesubtitle{Observability for inference servers}
Triton exposes a \textbf{Prometheus-compatible} metrics endpoint at \texttt{:8002/metrics}.
\vspace{2mm}
\begin{tabular}{@{}p{6.0cm}p{6.8cm}@{}}
\toprule
Metric & Meaning \\
\midrule
\texttt{nv\_inference\_request\_success} & Total successful inference requests \\
\texttt{nv\_inference\_queue\_duration\_us} & Time requests spend waiting in queue \\
\texttt{nv\_inference\_compute\_infer\_duration\_us} & Pure GPU compute time \\
\texttt{nv\_gpu\_utilization} & GPU utilization \% \\
\texttt{nv\_gpu\_memory\_used\_bytes} & GPU memory consumed \\
\bottomrule
\end{tabular}

\textbf{Computed from raw metrics}: RPS, P50/P95/P99 latency, avg batch size --- derive these in Grafana via PromQL.
\end{frame}


\begin{frame}[t,fragile]{Dynamic Batching}
\framesubtitle{Triton's killer feature for high throughput}
\begin{columns}[T,onlytextwidth]
    \begin{column}{0.54\textwidth}
        \textbf{Without dynamic batching:}
        \begin{itemize}
            \item Each HTTP request $\Rightarrow$ 1 GPU forward pass
            \item GPU is mostly idle between requests
            \item Low throughput under concurrent load
        \end{itemize}
        \vspace{3mm}
        \textbf{With dynamic batching:}
        \begin{itemize}
            \item Triton queues requests for up to $N$ $\mu$s
            \item Merges them into a single large batch
            \item 1 GPU forward pass serves many clients
        \end{itemize}
    \end{column}
    \begin{column}{0.44\textwidth}
        Add to \texttt{config.pbtxt}:
\begin{lstlisting}[basicstyle=\small\ttfamily]
dynamic_batching {
  max_queue_delay_microseconds:
    100
}
\end{lstlisting}
        Triton will merge up to \texttt{max\_batch\_size} requests within the delay window.
    \end{column}
\end{columns}
\end{frame}

\begin{frame}[t,fragile]{Instance Groups}
\framesubtitle{Running multiple model copies in parallel}
        Even with dynamic batching, a single model instance can become a bottleneck at very high RPS.
        \vspace{2mm}

        \textbf{Instance Groups} let Triton load $N$ copies of the model on the GPU simultaneously, each serving requests independently.

\dbox{For small models (MobileNet-level), \texttt{count: 2--4} often doubles throughput. For large models (ViT-Large), VRAM is the limit.}
\end{frame}

\begin{frame}[t]{Load Testing with Locust}
\framesubtitle{Simulating real user traffic}
\textbf{Locust} is a Python load-testing tool that spawns many virtual users, each running your task functions asynchronously.
\vspace{2mm}
\begin{itemize}
    \item Define user behavior in pure Python --- no XML, no DSL.
    \item Web UI shows RPS, response times, and failures in real time.
    \item Ramp users up gradually to find the saturation point.
    \item Use together with Grafana to see GPU utilization vs.\ RPS.
\end{itemize}
\end{frame}

\begin{frame}[t]{Summary: The Complete Pipeline}
\framesubtitle{What you will implement in the tutorial}
\begin{enumerate}
    \item \textbf{Export}: \texttt{torch.onnx.export} with \texttt{dynamic\_axes} + Softmax head
    \item \textbf{Deploy}: Triton in Docker, \texttt{config.pbtxt}, ONNX Runtime backend
    \item \textbf{Wrap}: FastAPI + \texttt{tritonclient} gRPC, TorchVision preprocessing
    \item \textbf{Optimize}: \texttt{trtexec --fp16}, switch to \texttt{tensorrt\_plan} platform
    \item \textbf{Monitor}: Prometheus scrapes \texttt{:8002}, Grafana dashboard for latency \& GPU
    \item \textbf{Tune}: Dynamic batching delay + instance group count
    \item \textbf{Verify}: Locust load test, compare RPS before/after each optimization
\end{enumerate}
\end{frame}
